97 // Animated loading spinner
98 static void showSpinner(int durationMs = 1000, const std::string& message = "Processing") {
99 const char spinner[] = {'|', '/', '-', '\\'};
100 auto start = std::chrono::steady_clock::now();
101 int i = 0;
102
103 hideCursor();
104 std::cout << YELLOW << message << " " << RESET;
105
106 while (std::chrono::duration_cast<std::chrono::milliseconds>(
107 std::chrono::steady_clock::now() - start).count() < durationMs) {
108 std::cout << "\b" << spinner[i] << std::flush;
109 i = (i + 1) % 4;
110 std::this_thread::sleep_for(std::chrono::milliseconds(100));
111 }
112 std::cout << "\b" << GREEN << "DONE" << RESET << std::endl;
113 showCursor();
114 }

116 // Table printing utility
117 static void printTable(const std::vector<std::vector<std::string>>& rows,
118 const std::vector<std::string>& headers = {}) {
119 if (rows.empty()) return;
120
121 // Calculate column widths
122 std::vector<size_t> colWidths(rows[0].size(), 0);
123 for (const auto& row : rows) {
124 for (size_t i = 0; i < row.size(); ++i) {
125 colWidths[i] = std::max(colWidths[i], row[i].length());
126 }
127 }
128
129 // Print headers if provided
130 if (!headers.empty()) {
131 std::cout << BOLD << CYAN;
132 for (size_t i = 0; i < headers.size(); ++i) {
133 std::cout << std::setw(colWidths[i] + 2) << std::left << headers[i];
134 }
135 std::cout << RESET << std::endl;
136 printDivider('-');
137 }
138
139 // Print rows
140 for (const auto& row : rows) {
141 for (size_t i = 0; i < row.size(); ++i) {
142 std::cout << std::setw(colWidths[i] + 2) << std::left << row[i];
143 }
144 std::cout << std::endl;
145 }
